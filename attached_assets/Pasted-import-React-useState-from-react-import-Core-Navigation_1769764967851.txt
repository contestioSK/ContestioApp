// --- SUB-COMPONENTS ---

const MenuRow = ({ icon: Icon, label, isActive, onClick, variant, isPremium }) => {
  // Logic for anchored icons (Light mode visibility fix)
  const getIconWrapperClasses = (v, active) => {
    if (!active) {
      // Light Mode: bg-slate-100 gives icon volume. Text-slate-600 gives contrast.
      // Dark Mode: bg-transparent is preferred for minimalism as contrast is naturally higher.
      return "text-slate-600 dark:text-slate-500 bg-slate-100 dark:bg-transparent transition-all";
    }
    
    switch (v) {
      case 'blue': return "bg-blue-500/10 text-blue-600 dark:text-blue-400";
      case 'pink': return "bg-pink-500/10 text-pink-600 dark:text-pink-400";
      case 'amber': return "bg-amber-500/10 text-amber-600 dark:text-amber-400";
      case 'inverse': return "bg-slate-900 dark:bg-slate-100 text-white dark:text-slate-900 shadow-md"; 
      default: return "bg-slate-100 text-slate-900";
    }
  };

  const getDotClass = (v) => {
    switch (v) {
      case 'blue': return "bg-blue-600";
      case 'pink': return "bg-pink-600";
      case 'amber': return "bg-amber-600";
      default: return "bg-transparent";
    }
  };

  return (
    <button
      onClick={onClick}
      className={cn(
        "group w-full flex items-center gap-3 px-3 py-2.5 rounded-xl transition-all duration-200",
        isActive 
          ? "bg-white dark:bg-slate-950 shadow-sm border border-slate-200 dark:border-slate-800" 
          : "hover:bg-white dark:hover:bg-slate-800/50"
      )}
    >
      <div className={cn(
        "h-9 w-9 rounded-lg flex items-center justify-center relative",
        getIconWrapperClasses(variant, isActive)
      )}>
        <Icon size={20} strokeWidth={ICON_CONFIG.strokeWidth} />
        
        {isPremium && (
           <div className="absolute -top-1 -right-1 bg-amber-100 dark:bg-amber-900 text-amber-600 dark:text-amber-400 rounded-full p-0.5 border-2 border-white dark:border-slate-950 shadow-sm">
             <Crown size={8} strokeWidth={ICON_CONFIG.premiumStrokeWidth} />
           </div>
        )}
      </div>

      <span className={cn(
        "text-sm font-medium transition-colors",
        isActive ? "text-slate-900 dark:text-white font-bold" : "text-slate-600 dark:text-slate-400 group-hover:text-slate-900 dark:group-hover:text-slate-200"
      )}>
        {label}
      </span>

      {isActive && (
        <div className={cn("ml-auto w-1.5 h-1.5 rounded-full transition-all", getDotClass(variant))} />
      )}
    </button>
  );
};
