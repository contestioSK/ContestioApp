import React, { useState, useEffect } from "react";
import { useForm, Controller } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
import * as z from "zod";
import { 
  Trophy, Users, Clock, X, Save, FolderOpen, 
  Trash2, Swords, Target, Weight, Star, Calendar, 
  Plus, MapPin, Anchor, ShieldCheck, ChevronDown, AlertCircle,
  Timer, Zap, Info
} from "lucide-react";
import { motion, AnimatePresence } from "framer-motion";

// --- VALIDAČNÁ SCHÉMA (Zod SuperRefine) ---
const createBattleSchema = z.object({
  name: z.string().min(1, "Názov operácie je povinný").max(100),
  mode: z.enum(["most_fish", "total_weight", "biggest_fish", "best_3_fish", "best_5_fish"]),
  tripType: z.enum(["new", "existing"]),
  tripId: z.string().optional(),
  location: z.string().optional(),
  startAt: z.string().min(1, "Dátum štartu je povinný"),
  endAt: z.string().min(1, "Dátum ukončenia je povinný"),
  minWeightKg: z.number().optional(),
  includeOnlyVerified: z.boolean().default(false),
}).superRefine((data, ctx) => {
  if (new Date(data.endAt) <= new Date(data.startAt)) {
    ctx.addIssue({
      code: z.ZodIssueCode.custom,
      message: "Koniec musí byť po začiatku",
      path: ["endAt"],
    });
  }
  if (data.tripType === "new" && !data.location) {
    ctx.addIssue({
      code: z.ZodIssueCode.custom,
      message: "Lokalita je povinná pri novej výprave",
      path: ["location"],
    });
  }
});

// --- UI KOMPONENTY ---

  <div className="flex gap-4 mb-8">

      {number}
    </div>
    <div>
      <h3 className="text-xs font-black uppercase tracking-[0.2em] text-white leading-none mb-1">{title}</h3>
      <p className="text-[10px] font-bold text-slate-500 uppercase tracking-widest">{subtitle}</p>
    </div>
  </div>
);

const InputField = React.forwardRef(({ label, error, icon: Icon, ...props }, ref) => (
  <div className="space-y-2">
    <label className="text-[10px] font-black uppercase tracking-[0.2em] text-slate-500 ml-1">{label}</label>
    <div className="relative">
      {Icon && <Icon className="absolute left-4 top-1/2 -translate-y-1/2 text-slate-600" size={16} />}
      <input 
        {...props} 
        ref={ref}
        className={`w-full bg-slate-900/50 border ${error ? 'border-red-500/50' : 'border-slate-800'} focus:border-orange-500/50 focus:ring-1 focus:ring-orange-500/20 rounded-xl ${Icon ? 'pl-12' : 'px-4'} py-3.5 text-sm text-white transition-all outline-none placeholder:text-slate-700`}
      />
    </div>
    {error && <p className="text-[10px] text-red-500 font-bold ml-1 flex items-center gap-1 mt-1"><AlertCircle size={10} /> {error.message}</p>}
  </div>
));
InputField.displayName = "InputField";

const ModeCard = ({ mode, isSelected, onClick }) => {
  const meta = {
    most_fish: { icon: Target, desc: "Kto uloví najviac kusov" },
    total_weight: { icon: Weight, desc: "Súčet váhy všetkých rýb" },
    biggest_fish: { icon: Trophy, desc: "Rozhoduje najťažší kus" },
    best_3_fish: { icon: Star, desc: "Súčet 3 najlepších úlovkov" },
    best_5_fish: { icon: Swords, desc: "Súčet 5 najlepších úlovkov" }
  };
  const Icon = meta[mode.value].icon;

  return (
    <button
      type="button"
      onClick={onClick}
      className={`group relative flex flex-col p-4 rounded-2xl border transition-all text-left ${
        isSelected 
        ? 'bg-orange-500/10 border-orange-500/50 ring-1 ring-orange-500/20' 
        : 'bg-slate-900/30 border-slate-800 hover:border-slate-600'
      }`}
    >
      <div className={`mb-4 p-2.5 rounded-xl inline-flex w-fit transition-colors ${isSelected ? 'bg-orange-500 text-white' : 'bg-slate-800 text-slate-400 group-hover:text-slate-200'}`}>
        <Icon size={20} />
      </div>
      <div className="text-[11px] font-black uppercase tracking-tight text-white mb-1">{mode.label}</div>
      <div className="text-[9px] font-bold text-slate-500 leading-tight uppercase tracking-tighter">{meta[mode.value].desc}</div>
      {isSelected && <motion.div layoutId="active-mode" className="absolute top-3 right-3 w-1.5 h-1.5 rounded-full bg-orange-500 shadow-[0_0_10px_#f97316]" />}
    </button>
  );
};
