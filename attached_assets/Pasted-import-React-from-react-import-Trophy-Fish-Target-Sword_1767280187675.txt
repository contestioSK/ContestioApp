// --- 1. TACTICAL SOLID ICON COMPONENT ---
// Čistý, statický dizajn bez zbytočných animácií
const TacticalIcon = ({ icon: Icon, tier, isLocked }: { icon: any, tier: string, isLocked?: boolean }) => {
  const tierThemes: any = {
    locked: {
      border: "border-slate-800",
      accent: "text-slate-800",
      bg: "bg-slate-950/40"
    },
    bronze: {
      border: "border-orange-500/40",
      accent: "text-orange-600",
      bg: "bg-orange-500/5"
    },
    silver: {
      border: "border-slate-400/40",
      accent: "text-slate-400",
      bg: "bg-slate-400/5"
    },
    gold: {
      border: "border-amber-500/40",
      accent: "text-amber-500",
      bg: "bg-amber-500/5"
    }
  };

  const theme = isLocked ? tierThemes.locked : tierThemes[tier];

  return (
    <div className="relative w-16 h-16 sm:w-20 sm:h-20 flex items-center justify-center shrink-0">
      
      {/* Octagon Frame - Čistá technická geometria */}
      <div className={`absolute inset-0 border-2 ${theme.border} ${theme.bg} transition-colors duration-300`}
           style={{ clipPath: "polygon(30% 0%, 70% 0%, 100% 30%, 100% 70%, 70% 100%, 30% 100%, 0% 70%, 0% 30%)" }} 
      />
      
      {/* Static Tech Markers */}
      <div className={`absolute inset-0 opacity-40 ${theme.accent} pointer-events-none`}>
        <div className="absolute top-1 left-1/2 -translate-x-1/2 w-4 h-[1px] bg-current" />
        <div className="absolute bottom-1 left-1/2 -translate-x-1/2 w-4 h-[1px] bg-current" />
        <div className="absolute left-1 top-1/2 -translate-y-1/2 h-4 w-[1px] bg-current" />
        <div className="absolute right-1 top-1/2 -translate-y-1/2 h-4 w-[1px] bg-current" />
      </div>

      {/* Main Icon */}
      <div className={`relative z-10 ${theme.accent}`}>
        {isLocked ? (
          <Lock className="w-5 h-5 opacity-40" />
        ) : (
          <Icon className="w-7 h-7 sm:w-8 sm:h-8 stroke-[1.5]" />
        )}
      </div>
    </div>
  );
};

// --- 3. UI COMPONENTS ---

const BadgeCard = ({ badge }: { badge: any }) => {
  let currentTier = "locked";
  let nextThreshold = badge.thresholds.bronze;
  
  if (badge.current >= badge.thresholds.gold) { currentTier = "gold"; nextThreshold = badge.thresholds.gold; }
  else if (badge.current >= badge.thresholds.silver) { currentTier = "silver"; nextThreshold = badge.thresholds.gold; }
  else if (badge.current >= badge.thresholds.bronze) { currentTier = "bronze"; nextThreshold = badge.thresholds.silver; }

  const progress = Math.min((badge.current / nextThreshold) * 100, 100);
  const meta = TIER_META[currentTier];

  return (
    <div className="bg-slate-900/40 border border-white/5 rounded-[2rem] p-6 flex flex-col sm:flex-row items-center sm:items-start gap-6 shadow-xl hover:bg-slate-900/60 transition-colors duration-300">
      <TacticalIcon icon={badge.icon} tier={currentTier} isLocked={currentTier === 'locked'} />
      
      <div className="flex-1 w-full text-center sm:text-left">
        <div className="mb-2">
          <h4 className="text-sm font-black uppercase italic text-white tracking-widest leading-none mb-2">
            {badge.name}
          </h4>
          <div className="flex items-center justify-center sm:justify-start gap-2">
            <span className={`text-[9px] font-black uppercase tracking-[0.2em] ${meta.color}`}>{meta.label}</span>
            <span className="text-[9px] font-bold text-slate-600 uppercase tracking-wider">{badge.current} / {nextThreshold} {badge.unit}</span>
          </div>
        </div>

        {/* Solid Progress Bar */}
        <div className="relative h-1 w-full bg-slate-950 rounded-full mt-4 overflow-hidden border border-white/5">
           <div 
             className={`h-full ${currentTier === 'locked' ? 'bg-slate-800' : 'bg-lime-500 opacity-80'}`}
             style={{ width: `${progress}%` }}
           />
        </div>
      </div>
    </div>
  );
};
