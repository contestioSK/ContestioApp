// --- TACTICAL ICON COMPONENT (LIGHT MODE) ---
// Optimalizované pre svetlé rozhranie so zachovaním industriálneho charakteru
const TacticalIcon = ({ icon: Icon, label, variant = "neutral" }: { 
  icon: any, 
  label?: string, 
  variant?: "neutral" | "active" | "action" | "danger" 
}) => {
  const themes = {
    // Svetlá neutrálna: jemné šedé tóny, ktoré nebijú do očí
    neutral: "text-slate-400 border-slate-200 bg-white shadow-sm",
    // Aktívna: sýtejšia azúrová pre jasný kontrast na bielom
    active: "text-cyan-600 border-cyan-200 bg-cyan-50 shadow-[0_0_15px_rgba(8,145,178,0.1)]",
    // Akčná: čierna/tmavá bridlica na bielom pôsobí najviac prémiovo
    action: "text-slate-900 border-slate-900 bg-white shadow-[0_4px_10px_rgba(0,0,0,0.05)]",
    // Nebezpečenstvo: upravená červená pre biely podklad
    danger: "text-red-600 border-red-200 bg-red-50"
  };

  return (
    <div className="flex flex-col items-center gap-3 group">
      <div className="relative w-14 h-14 flex items-center justify-center transition-all duration-300">
        {/* Octagon Frame - Svetlá verzia s výraznejším obrysom */}
        <div 
          className={`absolute inset-0 border-2 transition-transform duration-500 group-hover:rotate-45 ${themes[variant]}`}
          style={{ clipPath: "polygon(30% 0%, 70% 0%, 100% 30%, 100% 70%, 70% 100%, 30% 100%, 0% 70%, 0% 30%)" }} 
        />
        
        {/* Technical Markers (Corners) - Jemnejšie pre svetlý mód */}
        <div className={`absolute inset-0 opacity-40 ${themes[variant].split(' ')[0]} pointer-events-none`}>
           <div className="absolute top-1.5 left-1/2 -translate-x-1/2 w-px h-1 bg-current" />
           <div className="absolute bottom-1.5 left-1/2 -translate-x-1/2 w-px h-1 bg-current" />
        </div>

        {/* Icon - Mierne hrubší stroke pre lepšiu viditeľnosť */}
        <Icon className={`relative z-10 w-5 h-5 stroke-[2] transition-transform duration-300 group-hover:scale-110`} />
      </div>
      {label && (
        <span className="text-[8px] font-black uppercase tracking-[0.2em] text-slate-400 group-hover:text-slate-900 transition-colors text-center max-w-[80px] leading-tight">
          {label}
        </span>
      )}
    </div>
  );
};
