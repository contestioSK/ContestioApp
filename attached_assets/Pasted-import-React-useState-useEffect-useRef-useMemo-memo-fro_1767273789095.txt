// --- 2. LOGIC HOOK ---

const useBadgeGamification = (userBadges: any[], badgeProgress: Record<string, number>) => {
  const { unlockedSet, unlockedList } = useMemo(() => {
    const set = new Set(userBadges.map(b => `${b.badgeType}_${b.tier}`));
    const list: any[] = [];
    
    Object.values(BADGE_DEFINITIONS).forEach(def => {
      ['gold', 'silver', 'bronze'].forEach(tier => {
        if (set.has(`${def.id}_${tier}`)) {
          list.push({ ...def, tier, tierInfo: def.tiers[tier] });
        }
      });
    });
    
    return { unlockedSet: set, unlockedList: list };
  }, [userBadges]);

  const nextGoal = useMemo(() => {
    let bestCandidate: any = null;

    Object.values(BADGE_DEFINITIONS).forEach(badgeDef => {
      let targetTier = null;
      if (!unlockedSet.has(`${badgeDef.id}_bronze`)) targetTier = 'bronze';
      else if (!unlockedSet.has(`${badgeDef.id}_silver`)) targetTier = 'silver';
      else if (!unlockedSet.has(`${badgeDef.id}_gold`)) targetTier = 'gold';

      if (targetTier) {
        const currentValue = badgeProgress[badgeDef.id] || 0;
        const threshold = badgeDef.tiers[targetTier].threshold;
        const remaining = Math.max(0, threshold - currentValue);
        const percent = Math.min((currentValue / threshold) * 100, 100);

        if (!bestCandidate || percent > bestCandidate.percent) {
          bestCandidate = {
            badgeDef,
            tier: targetTier,
            currentValue,
            threshold,
            percent,
            remaining
          };
        }
      }
    });

    return bestCandidate;
  }, [unlockedSet, badgeProgress]);

  return { unlockedSet, unlockedList, nextGoal };
};
