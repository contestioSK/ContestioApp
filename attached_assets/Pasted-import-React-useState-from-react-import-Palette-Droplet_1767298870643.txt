const ColorCard = ({ color, isDark }: { color: any, isDark: boolean }) => {
  const [copied, setCopied] = useState(false);
  const hex = isDark ? color.dark : color.light;

  const handleCopy = () => {
    const el = document.createElement('textarea');
    el.value = hex;
    document.body.appendChild(el);
    el.select();
    document.execCommand('copy');
    document.body.removeChild(el);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div className={`p-6 rounded-[2.5rem] border transition-all duration-500 ${
      isDark ? 'bg-white/5 border-white/5 shadow-2xl backdrop-blur-md' : 'bg-white border-slate-200 shadow-xl'
    }`}>
      <div className="flex justify-between items-start mb-6">
        <div 
          className="w-14 h-14 rounded-2xl flex items-center justify-center shadow-lg transition-transform hover:scale-110"
          style={{ backgroundColor: hex, color: isDark ? '#0c1f28' : '#ffffff' }}
        >
          <color.icon className="w-6 h-6 stroke-[2.5]" />
        </div>
        <button onClick={handleCopy} className={`p-2 transition-opacity ${isDark ? 'text-slate-500 hover:text-white' : 'text-slate-400 hover:text-slate-900'}`}>
          {copied ? <Check className="w-4 h-4 text-emerald-500" /> : <Copy className="w-4 h-4" />}
        </button>
      </div>

      <h3 className={`text-xl font-black uppercase italic tracking-tighter mb-1 ${isDark ? 'text-white' : 'text-slate-900'}`}>
        {color.name}
      </h3>
      <p className={`text-[10px] font-bold uppercase tracking-widest leading-relaxed mb-6 h-8 overflow-hidden ${isDark ? 'text-slate-500' : 'text-slate-400'}`}>
        {color.usage}
      </p>

      <div className={`p-3 rounded-xl font-mono text-xs flex justify-between items-center ${isDark ? 'bg-slate-950/80 text-orange-400' : 'bg-slate-100 text-slate-600 border border-slate-200/50'}`}>
        <span>{hex}</span>
        <div className="w-3 h-3 rounded-full shadow-sm" style={{ backgroundColor: hex }} />
      </div>
    </div>
  );
};
