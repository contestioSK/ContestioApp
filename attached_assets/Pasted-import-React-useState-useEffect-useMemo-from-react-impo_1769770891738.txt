function CircularProgress({ value, max, size = 120, strokeWidth = 10, className = "", children, status = "normal" }) {
const percentage = Math.min((value / max) * 100, 100);
const radius = (size - strokeWidth) / 2;
const circumference = radius * 2 * Math.PI;
const strokeDashoffset = circumference - (percentage / 100) * circumference;

const colorClass = status === 'completed' ? 'text-emerald-500' : (status === 'near' ? 'text-amber-500' : (status === 'start' ? 'text-slate-300 dark:text-slate-700' : 'text-blue-600 dark:text-blue-500'));

return (
<div className={`relative ${className}`} style={{ width: size, height: size }}>
<svg className="transform -rotate-90" width={size} height={size}>
<circle
cx={size / 2} cy={size / 2} r={radius}
stroke="currentColor" strokeWidth={strokeWidth}
fill="transparent"
className="text-slate-100 dark:text-slate-800"
/>
<circle
cx={size / 2} cy={size / 2} r={radius}
stroke="currentColor" strokeWidth={strokeWidth}
fill="transparent"
strokeDasharray={`${circumference} ${circumference}`}
strokeDashoffset={strokeDashoffset}
strokeLinecap="round"
className={`${colorClass} transition-all duration-1000 ease-out`}
/>
</svg>
<div className="absolute inset-0 flex items-center justify-center flex-col">
{children}
</div>
</div>
);
}

const formatNumber = (val) => val.toLocaleString('sk-SK');

const getGoalNarrative = (goal) => {
const p = goal.percentage;
const remaining = formatNumber(goal.target - goal.current);
const unit = goal.unit;

// 1. STAV: COMPLETED (100%) - Hrdosť (Hero) / Pokora (List)
if (p >= 100) {
return {
status: 'completed',
badgeText: 'ZVLÁDNUTÉ',
badgeVariant: 'green',
headline: 'Dokázal si to.',
subtext: 'Cieľ je pokorený.',
colorClass: 'text-emerald-600 dark:text-emerald-400',
icon: Trophy,
progressColor: 'bg-emerald-500'
};
}

// 2. STAV: NEAR COMPLETION (80-99%) - Napätie
if (p >= 80) {
return {
status: 'near',
badgeText: 'FINÁLE BLÍZKO',
badgeVariant: 'gold',
headline: `Už len ${remaining} ${unit}.`, // Stručné
subtext: 'Toto je moment, ktorý rozhoduje.', // Úderné
colorClass: 'text-amber-600 dark:text-amber-400',
icon: Flame,
progressColor: 'bg-amber-500'
};
}

// 3. STAV: FLOW (20-79%) - Minimalistická sila (Variant C)
if (p >= 20) {
return {
status: 'normal',
badgeText: 'V TEMPE',
badgeVariant: 'blue',
headline: 'Drž tempo.',
subtext: 'Presne takto sa robí sezóna.', // Minimalistické
colorClass: 'text-blue-600 dark:text-blue-400',
icon: Zap,
progressColor: 'bg-blue-600'
};
}

// 4. STAV: START / STAGNÁCIA (0-19%)
return {
status: 'start',
badgeText: 'SEZÓNA SA ZAČÍNA',
badgeVariant: 'slate',
headline: 'Sezóna sa práve začína.',
subtext: 'Zapíš prvý úlovok a všetko sa pohne.',
colorClass: 'text-slate-500 dark:text-slate-400',
icon: Hourglass,
progressColor: 'bg-slate-300 dark:bg-slate-700'
};
};

const calculateProgress = (goal, catches) => {
let current = 0;
switch (goal.type) {
case 'total_weight': current = catches.reduce((acc, c) => acc + (c.weight || 0), 0); break;
case 'total_catches': current = catches.length; break;
case 'max_weight': current = catches.reduce((max, c) => Math.max(max, c.weight || 0), 0); break;
case 'species_variety': current = new Set(catches.map(c => c.species)).size; break;
case 'trips_count': current = new Set(catches.map(c => c.date.toDateString())).size; break;
case 'min_weight_catch_count':
const minW = parseFloat(goal.parameter) || 0;
current = catches.filter(c => (c.weight || 0) >= minW).length;
break;
case 'min_size_catch_count':
const minL = parseFloat(goal.parameter) || 0;
current = catches.filter(c => (c.length || 0) >= minL).length;
break;
case 'spot_catch_count':
const spot = (goal.parameter || '').toLowerCase();
current = catches.filter(c => (c.location || '').toLowerCase().includes(spot)).length;
break;
case 'bait_catch_count':
const bait = (goal.parameter || '').toLowerCase();
current = catches.filter(c => (c.bait || '').toLowerCase().includes(bait)).length;
break;
default: current = 0;
}

current = parseFloat(current.toFixed(1));
const target = parseFloat(goal.target);
const percentage = target > 0 ? Math.min((current / target) * 100, 100) : 0;

const baseGoal = { current, target, percentage, isCompleted: current >= target };
const narrative = getGoalNarrative({ ...goal, ...baseGoal });

return { ...baseGoal, ...narrative };
};
