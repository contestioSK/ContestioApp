// --- 2. LOGIC HOOK ---

const useBadgeGamification = (userBadges: any[], badgeProgress: Record<string, number>) => {
  const { unlockedSet, unlockedList } = useMemo(() => {
    const set = new Set(userBadges.map(b => `${b.badgeType}_${b.tier}`));
    const list: any[] = [];
    
    Object.values(BADGE_DEFINITIONS).forEach(def => {
      ['gold', 'silver', 'bronze'].forEach(tier => {
        if (set.has(`${def.id}_${tier}`)) {
          list.push({ ...def, tier, tierInfo: def.tiers[tier] });
        }
      });
    });
    
    return { unlockedSet: set, unlockedList: list };
  }, [userBadges]);

  const nextGoal = useMemo(() => {
    let bestCandidate: any = null;

    Object.values(BADGE_DEFINITIONS).forEach(badgeDef => {
      let targetTier = null;
      if (!unlockedSet.has(`${badgeDef.id}_bronze`)) targetTier = 'bronze';
      else if (!unlockedSet.has(`${badgeDef.id}_silver`)) targetTier = 'silver';
      else if (!unlockedSet.has(`${badgeDef.id}_gold`)) targetTier = 'gold';

      if (targetTier) {
        const currentValue = badgeProgress[badgeDef.id] || 0;
        const threshold = badgeDef.tiers[targetTier].threshold;
        const remaining = Math.max(0, threshold - currentValue);
        const percent = Math.min((currentValue / threshold) * 100, 100);

        if (!bestCandidate || percent > bestCandidate.percent) {
          bestCandidate = {
            badgeDef,
            tier: targetTier,
            currentValue,
            threshold,
            percent,
            remaining
          };
        }
      }
    });

    return bestCandidate;
  }, [unlockedSet, badgeProgress]);

  return { unlockedSet, unlockedList, nextGoal };
};

// --- VISUAL EFFECTS: CONFETTI ---
const ConfettiEffect = () => {
  // Generujeme fixný počet častíc s náhodnými parametrami pre CSS animáciu
  const particles = Array.from({ length: 60 }).map((_, i) => {
    const left = Math.random() * 100;
    const animDelay = Math.random() * 1.5; // Rýchlejší štart
    const animDuration = 2 + Math.random() * 2;
    const colors = ['#84cc16', '#f59e0b', '#a855f7', '#ec4899', '#3b82f6'];
    const color = colors[Math.floor(Math.random() * colors.length)];
    const size = 6 + Math.random() * 8;
    
    return (
      <div 
        key={i}
        className="absolute top-[-20px] rounded-sm pointer-events-none"
        style={{
          left: `${left}%`,
          width: `${size}px`,
          height: `${size * 0.6}px`,
          backgroundColor: color,
          opacity: 0.8,
          animation: `confetti-fall ${animDuration}s linear forwards`,
          animationDelay: `${animDelay}s`,
          transform: `rotate(${Math.random() * 360}deg)`
        }}
      />
    );
  });

  return (
    <div className="fixed inset-0 pointer-events-none z-[100] overflow-hidden">
      <style>{`
        @keyframes confetti-fall {
          0% { transform: translateY(-50px) rotate(0deg) translateX(0px); opacity: 1; }
          25% { transform: translateY(25vh) rotate(90deg) translateX(20px); }
          50% { transform: translateY(50vh) rotate(180deg) translateX(-20px); }
          75% { transform: translateY(75vh) rotate(270deg) translateX(20px); }
          100% { transform: translateY(110vh) rotate(360deg) translateX(0px); opacity: 0; }
        }
      `}</style>
      {particles}
    </div>
  );
};
