// --- EMPTY STATE COMPONENT ---
function EmptyState({ tab }: { tab: 'active' | 'history' }) {
  const isHistory = tab === 'history';

  return (
    <div className="flex flex-col items-center justify-center py-20 px-4 text-center rounded-[32px] border-2 border-dashed border-white/5 bg-white/[0.02] animate-in fade-in duration-500">
      <div className="w-24 h-24 bg-[#131b2e] rounded-full flex items-center justify-center mb-6 shadow-xl border border-white/5 relative group">
        {/* Glow effect */}
        <div className={`absolute inset-0 rounded-full blur-xl opacity-20 ${isHistory ? 'bg-purple-500' : 'bg-cyan-500'}`}></div>
        
        {isHistory ? (
           <Trophy size={40} className="text-purple-400 relative z-10" />
        ) : (
           <Compass size={40} className="text-cyan-400 relative z-10" />
        )}
      </div>

      <h3 className="text-2xl font-bold text-white mb-3">
        {isHistory ? 'História zíva prázdnotou' : 'Žiadne naplánované výpravy'}
      </h3>
      
      <p className="text-slate-400 max-w-sm mb-8 text-lg leading-relaxed">
        {isHistory 
          ? 'Tvoj denník je zatiaľ čistý. Všetky tvoje legendárne úlovky a spomienky sa uložia práve sem.' 
          : 'Voda volá! Naplánuj si termín, vyber revír a my ti postrážime počasie.'}
      </p>

      {!isHistory && (
        <button className="bg-cyan-500 hover:bg-cyan-400 text-white font-bold px-8 py-4 rounded-full shadow-lg shadow-cyan-900/20 transition-all transform hover:-translate-y-1 flex items-center gap-2">
          <Plus size={20} strokeWidth={3} />
          <span>Naplánovať prvú výpravu</span>
        </button>
      )}
    </div>
  );
}

// --- TRIP CARD COMPONENT ---
function TripCard({ trip }: { trip: any }) {
  if (trip.locked) {
    return (
      <div className="relative h-[280px] rounded-[32px] bg-[#131b2e] border border-white/5 overflow-hidden group cursor-pointer">
        <div className="absolute inset-0 bg-black/60 z-10 flex flex-col items-center justify-center text-center p-6 backdrop-blur-[2px]">
          <div className="w-12 h-12 bg-white/10 rounded-full flex items-center justify-center mb-4 text-white/50">
            <Lock size={24} />
          </div>
          <h3 className="text-slate-300 font-bold text-lg mb-1">{trip.name}</h3>
          <p className="text-slate-500 text-sm mb-4">Táto výprava je dostupná len v Premium verzii.</p>
          <button className="bg-amber-500/10 text-amber-500 border border-amber-500/20 px-4 py-2 rounded-lg text-xs font-bold uppercase tracking-wider hover:bg-amber-500 hover:text-white transition-all">
            Prejsť na Premium
          </button>
        </div>
        {trip.coverImageUrl && (
          <img src={trip.coverImageUrl} className="absolute inset-0 w-full h-full object-cover opacity-20 grayscale" alt="" />
        )}
      </div>
    );
  }

  return (
    <div className="relative h-[280px] rounded-[32px] overflow-hidden group cursor-pointer border border-white/10 shadow-2xl shadow-black/20">
      <div className="absolute inset-0 bg-[#131b2e]">
        {trip.coverImageUrl ? (
          <img 
            src={trip.coverImageUrl} 
            className="w-full h-full object-cover transition-transform duration-700 group-hover:scale-110" 
            alt={trip.name} 
          />
        ) : (
          <div className="w-full h-full bg-gradient-to-br from-cyan-900 to-blue-900 opacity-50" />
        )}
      </div>
      <div className="absolute inset-0 bg-gradient-to-t from-[#0B1120] via-[#0B1120]/60 to-transparent opacity-90 group-hover:opacity-80 transition-opacity" />

      <div className="absolute inset-0 p-6 flex flex-col justify-between z-10">
        <div className="flex justify-between items-start">
          <div className={`
            px-3 py-1 rounded-full text-[10px] font-bold uppercase tracking-wider border backdrop-blur-md shadow-lg
            ${trip.status === 'upcoming' 
              ? 'bg-blue-500/20 border-blue-400/30 text-blue-200' 
              : 'bg-emerald-500/20 border-emerald-400/30 text-emerald-200'}
          `}>
            {trip.status === 'upcoming' ? 'Naplánované' : 'Ukončené'}
          </div>
          <button className="w-8 h-8 rounded-full bg-black/20 backdrop-blur-md border border-white/10 flex items-center justify-center text-white hover:bg-white hover:text-black transition-all">
            <MoreHorizontal size={16} />
          </button>
        </div>

        <div>
          <h3 className="text-2xl font-bold text-white mb-2 leading-tight group-hover:translate-x-1 transition-transform">
            {trip.name}
          </h3>
          <div className="flex flex-col gap-2 mb-4">
            <div className="flex items-center gap-2 text-slate-300 text-sm">
              <MapPin size={14} className="text-cyan-400" />
              {trip.location}
            </div>
            <div className="flex items-center gap-2 text-slate-400 text-xs">
              <Calendar size={12} />
              {trip.startDate.toLocaleDateString()} - {trip.endDate.toLocaleDateString()}
            </div>
          </div>

          {(trip.catchCount > 0 || trip.biggestFish) && (
            <div className="flex gap-3 pt-4 border-t border-white/10">
              <div className="flex flex-col">
                <span className="text-[10px] text-slate-500 uppercase font-bold">Úlovkov</span>
                <span className="text-white font-bold">{trip.catchCount}</span>
              </div>
              {trip.biggestFish && (
                 <div className="flex flex-col">
                   <span className="text-[10px] text-slate-500 uppercase font-bold">Top Ryba</span>
                   <span className="text-amber-400 font-bold">{trip.biggestFish}</span>
                 </div>
              )}
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
