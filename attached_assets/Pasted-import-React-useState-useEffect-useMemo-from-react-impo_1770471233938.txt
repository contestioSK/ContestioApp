  const baseStyles = "inline-flex items-center justify-center rounded-xl font-bold transition-all active:scale-95 disabled:opacity-50";
  const variants = {
    primary: "bg-orange-600/10 border border-orange-500/50 text-orange-500 hover:bg-orange-600 hover:text-white hover:shadow-[0_0_20px_rgba(249,115,22,0.2)]",

  };
  const sizes = {
    sm: "px-3 py-1.5 text-[11px]",
    md: "px-5 py-2.5 text-xs",
    lg: "px-8 py-3.5 text-sm uppercase tracking-wider"
  };
  return (
    <button className={`${baseStyles} ${variants[variant]} ${sizes[size]} ${className}`} {...props}>

const HorizontalBarChart = ({ data }) => {
  const max = Math.max(...data.map(d => d.weight), 1);
  return (
    <div className="space-y-4">
      {data.map((d, i) => (
        <div key={i}>
          <div className="flex justify-between text-[11px] mb-1.5">
            <span className="text-slate-300 font-bold">{d.name}</span>
            <span className="text-orange-500 font-mono font-bold">{d.weight.toFixed(1)} kg</span>
          </div>
          <div className="h-2 bg-slate-950 rounded-full overflow-hidden border border-slate-900">
            <div 
              className={`h-full transition-all duration-1000 ease-out shadow-[0_0_10px_rgba(0,0,0,0.5)] ${d.color}`} 
              style={{ width: `${(d.weight / max) * 100}%` }}
            ></div>
          </div>
        </div>
      ))}
    </div>
  );
};
